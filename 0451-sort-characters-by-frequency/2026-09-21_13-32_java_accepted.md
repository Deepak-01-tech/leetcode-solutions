## Solution

```java
class Solution {
    public String frequencySort(String s) {
        HashMap<Character,Integer> map =new HashMap<>();
        for(char ch : s.toCharArray()){
            map.put(ch,map.getOrDefault(ch,0)+1);
        }
        String a="";
        while(!map.isEmpty()){
            char maxchar=0;
            int max=0;
            for(char ch : map.keySet()){
                if(map.get(ch)>max){
                    max=map.get(ch);
                    maxchar=ch;
                }
            }
            for(int i=0;i<max;i++){
                a+=maxchar;
            }
            map.remove(maxchar);
        }
        return a;
    }
}
```
