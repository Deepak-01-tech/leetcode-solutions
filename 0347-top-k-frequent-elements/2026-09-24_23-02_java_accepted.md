## Solution

```java
class Solution {
    public int[] topKFrequent(int[] nums, int k) {
        HashMap<Integer,Integer> map=new HashMap<>();
        for(int i : nums){
            map.put(i,map.getOrDefault(i,0)+1);
        }
        int index=0;
        int[] arr=new int[k];
        while(index<k){
            int max=0;
            int num=0;
            for(int key : map.keySet()){
                if(map.get(key)>max){
                    max=map.get(key);
                    num=key;
                }
            }
            arr[index]=num;
            map.remove(num);
            index++;

        }
        return arr;
        
    }
}
```
