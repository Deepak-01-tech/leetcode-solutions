## Solution

```java
class Solution {
    public int sumSubarrayMins(int[] arr) {
        int n = arr.length;
        long ans = 0;

        int[] stack = new int[n];
        int top = -1;

        for (int i = 0; i <= n; i++) {

            while (top >= 0 && (i == n || arr[stack[top]] > arr[i])) {
                int mid = stack[top--];

                int left;
                if (top == -1)
                    left = mid + 1;
                else
                    left = mid - stack[top];

                int right = i - mid;

                ans += (long) arr[mid] * left * right;
            }

            if (i < n) {
                stack[++top] = i;
            }
        }

        return (int)(ans % 1000000007);
    }
}
```
