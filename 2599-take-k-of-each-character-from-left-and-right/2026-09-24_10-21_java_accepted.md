## Solution

```java
class Solution {
    public int takeCharacters(String s, int k) {
        int[] arr=new int[3];
        for(int i=0;i<s.length();i++){
            char ch=s.charAt(i);
            arr[ch-'a']++;
        }
        if(arr[0]<k || arr[1]<k || arr[2]<k){
            return -1;
        }
        int l=0;
        int max=0;
        int [] w=new int[3];
        for(int r=0;r<s.length();r++){
            w[s.charAt(r)-'a']++;
            while(w[0]>arr[0]-k || w[1]>arr[1]-k || w[2]>arr[2]-k){
                w[s.charAt(l)-'a']--;
                l++;

            }
                max=Math.max(max,r-l+1);

        }


        return s.length()-max;
        
    }
}
```
